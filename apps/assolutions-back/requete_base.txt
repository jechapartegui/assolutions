CREATE TABLE IF NOT EXISTS public.contacts
(
    id integer NOT NULL GENERATED ALWAYS AS IDENTITY ( INCREMENT 1 START 1 MINVALUE 1 MAXVALUE 2147483647 CACHE 1 ),
    object_type character varying(50) COLLATE pg_catalog."default" NOT NULL,
    object_id integer NOT NULL,
    contact_type character varying(50) COLLATE pg_catalog."default" NOT NULL,
    contact_value character varying COLLATE pg_catalog."default",
    diffusion boolean,
    contact_list character varying COLLATE pg_catalog."default" NOT NULL DEFAULT 'liste_contact'::character varying,
    info character varying COLLATE pg_catalog."default",
    pref boolean NOT NULL,
    CONSTRAINT contacts_pkey PRIMARY KEY (id)
)

TABLESPACE pg_default;

    WHEN p.contacts IS NULL THEN '[]'::jsonb

    -- si contacts est du texte JSON
    ELSE p.contacts::jsonb
  END
) elem
WHERE (elem->>'Value') IS NOT NULL
  AND (elem->>'Value') <> '';

CREATE TABLE IF NOT EXISTS public.login_project
(
    id integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    login_id integer NOT NULL,
    project_id integer NOT NULL,
    date_creation timestamp with time zone NOT NULL DEFAULT now(),

    CONSTRAINT fk_login_project_compte
        FOREIGN KEY (login_id)
        REFERENCES public.compte (id)
        ON DELETE CASCADE,

    CONSTRAINT fk_login_project_project
        FOREIGN KEY (project_id)
        REFERENCES public.project (id)
        ON DELETE CASCADE,

    CONSTRAINT uq_login_project_login_project
        UNIQUE (login_id, project_id)
);

CREATE INDEX IF NOT EXISTS idx_login_project_login_id
    ON public.login_project (login_id);

CREATE INDEX IF NOT EXISTS idx_login_project_project_id
    ON public.login_project (project_id);

    INSERT INTO public.login_project (login_id, project_id)
SELECT DISTINCT
    p.compte AS login_id,
    s.project_id AS project_id
FROM public.inscription_saison ins
INNER JOIN public.personne p
    ON p.id = ins.personne_id
INNER JOIN public.saison s
    ON s.id = ins.saison_id
WHERE p.compte IS NOT NULL
  AND s.project_id IS NOT NULL
  AND ins.active = true
ON CONFLICT (login_id, project_id) DO NOTHING;

  parent_id integer NULL,

  recette boolean NOT NULL,
  actif boolean NOT NULL DEFAULT true,
  ordre integer NOT NULL DEFAULT 0
);

ON classe_comptable(project_id);

CREATE TABLE IF NOT EXISTS budget_scenario (
  id serial PRIMARY KEY,
  project_id integer NOT NULL,
  saison_id integer NOT NULL,
  nom varchar(255) NOT NULL,
  scenario_defaut boolean NOT NULL DEFAULT false,
  info text NULL
);

CREATE INDEX IF NOT EXISTS idx_budget_scenario_project_saison
ON budget_scenario(project_id, saison_id);

CREATE TABLE IF NOT EXISTS budget_ligne (
  id serial PRIMARY KEY,
  budget_scenario_id integer NOT NULL,
  classe_comptable_id integer NOT NULL,
  montant_budget double precision NOT NULL,
  info text NULL,
  CONSTRAINT fk_budget_ligne_scenario
    FOREIGN KEY (budget_scenario_id)
    REFERENCES budget_scenario(id)
    ON DELETE CASCADE,
  CONSTRAINT fk_budget_ligne_classe
    FOREIGN KEY (classe_comptable_id)
    REFERENCES classe_comptable(id)
);

CREATE INDEX IF NOT EXISTS idx_budget_ligne_scenario
ON budget_ligne(budget_scenario_id);

CREATE INDEX IF NOT EXISTS idx_budget_ligne_classe
ON budget_ligne(classe_comptable_id);
