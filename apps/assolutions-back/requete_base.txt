CREATE TABLE IF NOT EXISTS public.login_project
(
    id integer GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    login_id integer NOT NULL,
    project_id integer NOT NULL,
    date_creation timestamp with time zone NOT NULL DEFAULT now(),

    CONSTRAINT fk_login_project_compte
        FOREIGN KEY (login_id)
        REFERENCES public.compte (id)
        ON DELETE CASCADE,

    CONSTRAINT fk_login_project_project
        FOREIGN KEY (project_id)
        REFERENCES public.project (id)
        ON DELETE CASCADE,

    CONSTRAINT uq_login_project_login_project
        UNIQUE (login_id, project_id)
);

CREATE INDEX IF NOT EXISTS idx_login_project_login_id
    ON public.login_project (login_id);

CREATE INDEX IF NOT EXISTS idx_login_project_project_id
    ON public.login_project (project_id);

    INSERT INTO public.login_project (login_id, project_id)
SELECT DISTINCT
    p.compte AS login_id,
    s.project_id AS project_id
FROM public.inscription_saison ins
INNER JOIN public.personne p
    ON p.id = ins.personne_id
INNER JOIN public.saison s
    ON s.id = ins.saison_id
WHERE p.compte IS NOT NULL
  AND s.project_id IS NOT NULL
  AND ins.active = true
